r18-scaleaux-on|10 -2500 0 0|meshes/sA-e3.msh|--hypre-method 8 --tol 1e-8 --maxit 3000 --solver-line "AMS Scale Auxiliary = Logical True"|ELMER=$HOME/temper-fem/speedup/elmer-r18
